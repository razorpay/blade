# ─ Border

> These tokens should be used along with the [makeBorderSize util](?path=/story/utils-makebordersize--docs).

export const Border = () => {
  const { theme } = useTheme();
  return (
    <>
      <br />
      <h3>Border Radius</h3>
      <table>
        <tbody>
          {Object.entries(theme.border.radius).map(([key, value]) => (
            <tr key={key}>
              <td>{`theme.border.radius.${key}`}</td>
              <td>
                {key === 'round' ? (
                  <div
                    style={{
                      width: '64px',
                      height: '64px',
                      backgroundColor: 'hsla(220, 30%, 96%, 1)',
                      borderRadius: `${makeBorderSize(value)}`,
                      border: '1px solid hsla(216, 15%, 54%, 0.18)',
                    }}
                  />
                ) : (
                  <div
                    style={{
                      width: '96px',
                      height: '64px',
                      backgroundColor: 'hsla(220, 30%, 96%, 1)',
                      borderRadius: `${makeBorderSize(value)}`,
                      border: '1px solid hsla(216, 15%, 54%, 0.18)',
                    }}
                  />
                )}
              </td>
            </tr>
          ))}
        </tbody>
      </table>
      <br />
      <h3>Border Width</h3>
      <table>
        <tbody>
          {Object.entries(theme.border.width).map(([key, value]) => (
            <tr key={key}>
              <td>{`theme.border.width.${key}`}</td>
              <td>
                <div
                  style={{
                    width: '64px',
                    height: '64px',
                    backgroundColor: 'hsla(220, 30%, 96%, 1)',
                    borderRadius: `${makeBorderSize(theme.border.radius.medium)}`,
                    border: `${value}px solid hsla(216, 15%, 54%, 0.18)`,
                  }}
                />
              </td>
            </tr>
          ))}
        </tbody>
      </table>
      <br />
      <h3>Border Dash</h3>
      <p>
        Dash patterns as [dash, gap] lengths. Use them where a CSS <code>border-style</code> keyword
        cannot set the pattern, for example an SVG <code>stroke-dasharray</code>:{' '}
        <code>{"theme.border.dash.dashed.join(' ')"}</code>.
      </p>
      <table>
        <tbody>
          {Object.entries(theme.border.dash).map(([key, value]) => (
            <tr key={key}>
              <td>{`theme.border.dash.${key}`}</td>
              <td>{`[${value.join(', ')}]`}</td>
              <td>
                <svg width="96" height="8" aria-hidden="true">
                  <line
                    x1="0"
                    x2="96"
                    y1="4"
                    y2="4"
                    stroke={theme.colors.surface.border.gray.normal}
                    strokeWidth={theme.border.width.thicker}
                    strokeDasharray={value.join(' ')}
                  />
                </svg>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </>
  );
};

<Border />
